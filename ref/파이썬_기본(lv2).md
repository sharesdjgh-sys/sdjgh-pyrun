# 0.자료형_6.집합 -------------------------------------------
# 집합(set)은 중복을 허용하지 않는 순서 없는 자료형을 의미한다.

# 집합 정의
s2 = set([1, 2, 3])
print(s2)  # {1, 2, 3}
print('name' in s2)  # False, 'name'은 s2 집합에 없음

# 교집합, 합집합, 차집합 구하기
s1 = set([1, 2, 3, 4, 5, 6])
s2 = set([4, 5, 6, 7, 8, 9])
print(s1 & s2)  # 교집합: {4, 5, 6}
print(s1 | s2)  # 합집합: {1, 2, 3, 4, 5, 6, 7, 8, 9}
print(s1 - s2)  # 차집합: {1, 2, 3}

print(s1.intersection(s2)) # s1과 s2의 교집합을 intersection 함수로 구함
print(s1.union(s2)) # s1과 s2의 합집합을 union 함수로 구함
print(s1.difference(s2)) # s1에서 s2를 뺀 차집합을 difference 함수로 구함

# 집합 관련 자체 함수
s1 = set([1, 2, 3])
s1.add(4)  # {1, 2, 3, 4} (add 함수로 값 한 개 추가하기)
print(s1)
s1.update([5, 6])  # {1, 2, 3, 4, 5, 6} (update 함수로 값 여러 개 추가하기)
print(s1)
s1.remove(2)  # {1, 3, 4, 5, 6} (remove 함수로 특정 값 제거하기)
print(s1)

class FourCal:  # FourCal 클래스를 정의
    def add(self, a, b):  # 덧셈 메서드
        return a + b  # 두 수를 더한 값을 반환

    def subtract(self, a, b):  # 뺄셈 메서드
        return a - b  # 두 수를 뺀 값을 반환

    def multiply(self, a, b):  # 곱셈 메서드
        return a * b  # 두 수를 곱한 값을 반환

    def divide(self, a, b):  # 나눗셈 메서드
        if b == 0:  # b가 0인지 확인
            return "Error: Division by zero"  # 0으로 나눌 경우 에러 메시지 반환
        return a / b  # 두 수를 나눈 값을 반환

# 객체에 first, second와 같은 초깃값을 설정해야 할 필요가 있을 때 자동으로 호출되는 메서드

class FourCal3:
    def setdata(self, first, second):  # self를 매개변수로 줌으로써 객체를 설정한다.
        self.first = first  # 첫 번째 숫자를 객체 변수 first에 저장
        self.second = second  # 두 번째 숫자를 객체 변수 second에 저장

    def add(self):
        result = self.first + self.second  # add 함수는 first + second를 반환한다.
        return result  # 결과를 반환

    def mul(self):
        result = self.first * self.second  # mul 함수는 first * second를 반환한다.
        return result  # 결과를 반환

    def sub(self):
        result = self.first - self.second  # sub 함수는 first - second를 반환한다.
        return result  # 결과를 반환

    def div(self):
        if self.second == 0:  # 0으로 나누기 방지
            return "Cannot divide by zero"  # 에러 메시지 반환
        result = self.first / self.second  # div 함수는 first / second를 반환한다.
        return result  # 결과를 반환

# 이번엔 생성자 __init__ 을 사용해 객체 생성과 동시에 변수를 설정해 봅시다.
class FourCal4:
    def __init__(self, first, second):  # 생성자 __init__은 객체 생성 시점에 바로 객체 변수를 설정한다.
        self.one = first  # 객체변수 one에 매개변수 first를 할당한다.
        self.two = second  # 객체변수 two에 매개변수 second를 할당한다.

    def add(self):
        result = self.one + self.two  # add 함수는 one + two를 반환한다.
        return result  # 결과를 반환

    def mul(self):
        result = self.one * self.two  # mul 함수는 one * two를 반환한다.
        return result  # 결과를 반환

    def sub(self):
        result = self.one - self.two  # sub 함수는 one - two를 반환한다.
        return result  # 결과를 반환

    def div(self):
        if self.two == 0:  # 0으로 나누기 방지
            return "Cannot divide by zero"  # 에러 메시지 반환
        result = self.one / self.two  # div 함수는 one / two를 반환한다.
        return result  # 결과를 반환

print("a.one:", a.one)  # 객체 a의 one 값 출력
print("a.two:", a.two)  # 객체 a의 two 값 출력

# 3.클래스_5)클래스의 상속

# 보통 상속은 기존 클래스를 변경하지 않고 기능을 추가하거나 기존 기능을 변경하려고 할 때 사용한다.

class MoreFourCal(FourCal4):  # FourCal 클래스를 상속하는 MoreFourCal 클래스를 만든다.
    def pow(self): # pow 메서드를 추가한다.
        result = self.one ** self.two # pow 메서드는 one ** two를 반환한다.
        return result

    def div(self):
        if self.two == 0:  # 나누는 값이 0인 경우 0을 리턴하도록 수정

class Engine:

"""### 🗂️ 미니 프로젝트: 바탕화면/다운로드 폴더 정리기

파이썬의 `os`와 `shutil` 모듈을 사용하면 확장자에 따라 파일을 자동으로 폴더에 분류해 넣을 수 있습니다.

**Step 1: 먼저 실습을 위한 가짜 파일들을 만들어 봅시다.**
"""

import os

# 실습용 폴더 만들기
base_dir = './정리안된_폴더'
if not os.path.exists(base_dir):
    os.makedirs(base_dir)

# 텍스트 파일과 이미지 파일 무작위 생성
file_names = ['메모1.txt', '사진1.jpg', '숙제.txt', '풍경.jpg', '아이디어.txt']

for name in file_names:
    file_path = os.path.join(base_dir, name)
    # 빈 파일 생성하기
    with open(file_path, 'w') as f:
        f.write("테스트용 파일입니다.")

print(f"'{base_dir}' 폴더에 실습용 파일들이 생성되었습니다. 좌측 폴더 탭에서 확인해보세요!")

import shutil

# 이동할 목적지 폴더 만들기
txt_dir = os.path.join(base_dir, '텍스트파일')
jpg_dir = os.path.join(base_dir, '이미지파일')

os.makedirs(txt_dir, exist_ok=True)
os.makedirs(jpg_dir, exist_ok=True)

# '정리안된_폴더' 안의 모든 파일 목록 가져오기
file_list = os.listdir(base_dir)

for file in file_list:
    # 파일명과 확장자 분리 (예: '메모1', '.txt')
    name, ext = os.path.splitext(file)

    # 이동할 원래 파일 경로
    source = os.path.join(base_dir, file)

    # 폴더는 이동하지 않고 건너뜀
    if os.path.isdir(source):
        continue

    # 확장자에 따라 파일 이동
    if ext == '.txt':
        shutil.move(source, os.path.join(txt_dir, file))
        print(f"[{file}] -> 텍스트파일 폴더로 이동 완료!")
    elif ext == '.jpg':
        shutil.move(source, os.path.join(jpg_dir, file))
        print(f"[{file}] -> 이미지파일 폴더로 이동 완료!")

print("\n✨ 폴더 정리가 모두 완료되었습니다!")

"""### Boolean Indexing(조건에 맞는 데이터만 '필터') 기법"""

scores = np.array([85, 42, 90, 77, 60])

# 80점 이상인 데이터만 추출
high_scores = scores[scores >= 80]
print(high_scores) # [85, 90]

# 조건에 맞는 데이터를 한 번에 수정하기
scores[scores < 60] = 0 # 60점 미만은 모두 0점으로 처리
print(scores) # [85, 0, 90, 77, 60]

# 만약 numpy가 아닌 list로 처리한다면?
for i in range(len(scores)):
    if scores[i] < 60:
        scores[i] = 0

print(scores)  # [85, 0, 90, 77, 60]

"""### 집계 함수와 Axis"""

arr = np.array([[1, 2], [3, 4]])
arr

print(np.sum(arr))          # 전체 합계: 10
print(np.sum(arr, axis=0))  # 열별 합계(세로): [4, 6]
print(np.sum(arr, axis=1))  # 행별 합계(가로): [3, 7]

# 조건 필터링 비교 - 파이썬 리스트 방식(반복문+조건문)
math_scores = [85, 42, 90, 77, 60]
sci_scores = [95, 30, 88, 92, 45]

# 수학 > 80 그리고 과학 > 80인 학생의 '수학 점수'만 따로 모으기
filtered_list = []
for m, s in zip(math_scores, sci_scores):
    if m >= 80 and s >= 80:
        filtered_list.append(m)

print(f"리스트 결과: {filtered_list}") # [85, 90]

# 조건 필터링 비교 - numpy 방식(논리 연산자 활용)
import numpy as np

math_np = np.array([85, 42, 90, 77, 60])
sci_np = np.array([95, 30, 88, 92, 45])

# 수학 > 80 그리고 과학 > 80인 학생의 '수학 점수'만 따로 모으기 (괄호로 묶어주는 것이 중요합니다!)
mask = (math_np >= 80) & (sci_np >= 80)
# mask = (math_np > 80) | (sci_np > 80)  이건  '수학 > 80' 또는 '과학 > 80' 인 학생
# mask = ~(math_np > 80) | ~(sci_np > 80) 이건  '수학 > 80' 이 아니거나 '과학 > 80' 이 아닌 학생


# 조건에 맞는 수학 점수만 추출
filtered_np = math_np[mask]

print(f"NumPy 결과: {filtered_np}") # [85, 90]
